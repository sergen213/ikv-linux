if(IKV_BUILD_CLIENT)
    find_package(SDL3 3.4 CONFIG REQUIRED)
    find_package(OpenGL REQUIRED)

    endif()
    find_package(Freetype REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(Fontconfig REQUIRED IMPORTED_TARGET fontconfig)
    pkg_check_modules(TurboJPEG REQUIRED IMPORTED_TARGET libturbojpeg) # original ui/loading*.jpg

    target_link_libraries(ikv-renderers PUBLIC ikv-scene ikv-protocol SDL3::SDL3 OpenGL::GL PRIVATE PkgConfig::TurboJPEG)
    target_compile_definitions(ikv-renderers PUBLIC IKV_HAS_VULKAN=${IKV_HAS_VULKAN})
    if(IKV_BUILD_VULKAN)
        target_sources(ikv-renderers PRIVATE src/renderer_vk.cpp src/vulkan_rt.cpp)
        target_link_libraries(ikv-renderers PUBLIC Vulkan::Vulkan)
        target_compile_definitions(ikv-renderers PRIVATE IKV_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders/vulkan")
        set(ikv_vulkan_shaders scene.vert scene.frag scene_rt.frag scene_rt_guided.frag shadow.vert shadow.frag fullscreen.vert temporal.frag post.frag overlay.frag particle.vert particle.frag contour.vert contour.frag glow.frag)
        set(ikv_spirv)
        foreach(shader IN LISTS ikv_vulkan_shaders)
            set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/vulkan/${shader}.spv")
            set(shader_includes)
            if(shader MATCHES "^(scene|shadow|particle|contour)\\.vert$" OR shader MATCHES "^scene.*\\.frag$" OR
               shader STREQUAL "post.frag" OR shader STREQUAL "overlay.frag" OR shader STREQUAL "temporal.frag")
                list(APPEND shader_includes "${CMAKE_CURRENT_SOURCE_DIR}/shaders/vulkan/frame.glsl")
            endif()
            if(shader STREQUAL "scene.frag" OR shader STREQUAL "scene_rt.frag" OR shader STREQUAL "scene_rt_guided.frag")
                list(APPEND shader_includes "${CMAKE_CURRENT_SOURCE_DIR}/shaders/vulkan/lighting.glsl")
            endif()
            if(shader STREQUAL "scene_rt.frag" OR shader STREQUAL "scene_rt_guided.frag")
                list(APPEND shader_includes "${CMAKE_CURRENT_SOURCE_DIR}/shaders/vulkan/rt_lighting.glsl")
            endif()
            if(shader STREQUAL "scene_rt_guided.frag" OR shader STREQUAL "temporal.frag")
                list(APPEND shader_includes "${CMAKE_CURRENT_SOURCE_DIR}/shaders/vulkan/rt_guide.glsl")
            endif()
            add_custom_command(OUTPUT "${output}"
                COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders/vulkan"
                COMMAND "${IKV_GLSLC}" --target-env=vulkan1.2 -I "${CMAKE_CURRENT_SOURCE_DIR}/shaders/vulkan"
                    "${CMAKE_CURRENT_SOURCE_DIR}/shaders/vulkan/${shader}" -o "${output}"
                DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/vulkan/${shader}" ${shader_includes}
                VERBATIM)
            list(APPEND ikv_spirv "${output}")
        endforeach()
        add_custom_target(ikv-shaders DEPENDS ${ikv_spirv})
        add_dependencies(ikv-renderers ikv-shaders)
        install(FILES ${ikv_spirv} DESTINATION "${CMAKE_INSTALL_DATADIR}/ikv/shaders/vulkan")
        # AMD FidelityFX SDK v1.1.4 / FSR 3.1.4 upscaler, frame interpolation, optical flow + Vulkan backend (MIT;
        # third_party/FidelityFX-SDK/IKV-PROVENANCE.txt). Frame generation pacing/presentation is IKV's (renderer_vk.cpp).
        # tools/ffx_fsr3_blobs.py replaces the SDK's Windows-only FidelityFX_SC: glslc SPIR-V + descriptor reflection.
        find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
        set(ikv_ffx "${CMAKE_CURRENT_SOURCE_DIR}/third_party/FidelityFX-SDK")
        set(ikv_ffx_blobs "${CMAKE_CURRENT_BINARY_DIR}/generated/ikv_fsr3_blobs.h")
        file(GLOB_RECURSE ikv_ffx_gpu CONFIGURE_DEPENDS "${ikv_ffx}/sdk/include/FidelityFX/gpu/*.h"
            "${ikv_ffx}/sdk/src/backends/vk/shaders/*.glsl")
        add_custom_command(OUTPUT "${ikv_ffx_blobs}"
            COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/generated"
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/ffx_fsr3_blobs.py" --glslc "${IKV_GLSLC}"
                --sdk "${ikv_ffx}" --output "${ikv_ffx_blobs}"
            DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/ffx_fsr3_blobs.py" ${ikv_ffx_gpu}
            VERBATIM)
        set(ikv_ffx_sources "${ikv_ffx}/sdk/src/components/fsr3upscaler/ffx_fsr3upscaler.cpp"
            "${ikv_ffx}/sdk/src/components/frameinterpolation/ffx_frameinterpolation.cpp"
            "${ikv_ffx}/sdk/src/components/opticalflow/ffx_opticalflow.cpp"
            "${ikv_ffx}/sdk/src/backends/vk/ffx_vk.cpp" "${ikv_ffx}/sdk/src/shared/ffx_assert.cpp"
            "${ikv_ffx}/sdk/src/shared/ffx_message.cpp" "${ikv_ffx}/sdk/src/shared/ffx_object_management.cpp"
            "${ikv_ffx}/sdk/src/shared/ffx_breadcrumbs_list.cpp")
        add_library(ikv-fsr3 STATIC ${ikv_ffx_sources} src/fsr_vk.cpp "${ikv_ffx_blobs}")
        set_source_files_properties(${ikv_ffx_sources} PROPERTIES COMPILE_OPTIONS "-w") # upstream sources (local patches: IKV-PROVENANCE.txt)
        target_include_directories(ikv-fsr3 SYSTEM PRIVATE "${ikv_ffx}/sdk/include" "${ikv_ffx}/sdk/src/shared"
            "${ikv_ffx}/sdk/src/backends/shared" "${ikv_ffx}/sdk/src/components" "${CMAKE_CURRENT_BINARY_DIR}/generated")
        target_include_directories(ikv-fsr3 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
        target_compile_options(ikv-fsr3 PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ikv-ffx-posix/ikv_ffx_posix.h")
        target_compile_features(ikv-fsr3 PUBLIC cxx_std_20)
        target_link_libraries(ikv-fsr3 PUBLIC Vulkan::Vulkan)
        target_link_libraries(ikv-renderers PRIVATE ikv-fsr3)
        list(APPEND ikv_targets ikv-fsr3)
        install(FILES "${ikv_ffx}/LICENSE.txt" DESTINATION "${CMAKE_INSTALL_DOCDIR}/third_party" RENAME FidelityFX-SDK-LICENSE.txt)
    endif()

foreach(target IN LISTS ikv_targets)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(IKV_ENABLE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    elseif(IKV_ENABLE_SANITIZERS)
        message(FATAL_ERROR "IKV_ENABLE_SANITIZERS requires GCC or Clang.")
    endif()
endforeach()
